USE practice_db;

CREATE TABLE teachers (
    teacher_id INT PRIMARY KEY,
    teacher_name VARCHAR(50),
    dept_name VARCHAR(50)
);

FROM departments d

FROM departments d
LEFT JOIN teachers t
ON d.dept_name = t.dept_name
WHERE t.teacher_name IS NULL;
